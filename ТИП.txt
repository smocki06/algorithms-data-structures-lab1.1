#include <iostream>
using namespace std; // нужна, чтобы не писать каждый раз std

int main() {
    // Размеры массивов
    int a = 5;
    int b = 4;

    // Исходные массивы (уже отсортированы по возрастанию)
    int A[] = {1, 3, 5, 7, 9};
    int B[] = {2, 4, 6, 8};

    // Массив для результата: в нём будет ровно a + b элементов
    int result[a + b];

    // индексы на текущие элементы в каждом массиве
    int i = 0; // смотрим на A[i]
    int j = 0; // смотрим на B[j]

    // цикл выполняется ровно a + b итераций
    for (int k = 0; k < a + b; ++k) {
        // Если массив A закончился (i >= a), то берём из B.
        // Или если массив B закончился (j >= b), то берём из A.
        // Или если оба ещё есть, то берём тот элемент, который меньше.

        if (i >= a) {
            // A уже весь использован — остаётся только B
            result[k] = B[j];  //Мы берём текущий элемент из массива B (на позиции j) и записываем его в результат на позицию k
            j++;
        }
        else if (j >= b) {
            // B уже весь использован — остаётся только A
            result[k] = A[i];
            i++;
        }
        else {
            // Оба массива ещё не кончились — сравниваем текущие элементы
            if (A[i] <= B[j]) {
                // Элемент из A меньше или равен — берём его
                result[k] = A[i];
                i++;
            }
            else {
                // Иначе берём элемент из B
                result[k] = B[j];
                j++;
            }
        }
    }

    // Вывод результата
    cout << "Объединённый массив: ";
    for (int k = 0; k < a + b; ++k) {
        cout << result[k] << " ";  // выводим элемент массива result с индексом k, а потом у нас будет пробел, чтобы числа не шли слитно
    }
    cout << endl;

    return 0;
}
